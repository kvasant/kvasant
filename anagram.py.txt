import itertools
import unittest
import threading


class Anagrams:

    global words
    words=dict()
    def __init__(self):
# words is made as a Singleton variable due to its high memory requirement
        global words
        fileLock=threading.Lock()
        if (len(words) == 0 ):    
           print('Loading words dictionary from large file: words.txt')
           with fileLock:
              with open('words.txt','r') as file:
                 words={ line.rstrip("\n").lower(): True  for line in file.readlines() }

    def isValid(self,w):
      if w in words:
        return True
      else:
        return False

    def get_anagrams(self, word):
        word=word.lower()
        loopLock=threading.Lock()
        with loopLock:
           answer=[]
           for wordPerm in set(itertools.permutations(word,len(word))):
             testWord=''.join(wordPerm)
             if self.isValid(testWord):
               answer.append(testWord)
        return (sorted(answer))


class TestAnagrams(unittest.TestCase):

        self.assertEqual(anagrams.get_anagrams('reserve'), ['reserve', 'reveres', 'reverse', 'severer'])
        self.assertEqual(anagrams.get_anagrams('three'), ['ether', 'there', 'three'])

        self.assertEqual(anagrams.get_anagrams(''), [])
